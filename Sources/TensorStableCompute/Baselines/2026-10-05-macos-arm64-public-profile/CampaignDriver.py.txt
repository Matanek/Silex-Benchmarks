from pathlib import Path
from datetime import datetime, timezone
import subprocess, json, hashlib, sys
root=Path.cwd()
bench=root/'Silex-Benchmarks'
res=bench/'Sources/TensorStableCompute/Baselines/2026-10-05-macos-arm64-public-profile'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def cmd(a):return subprocess.run(a,capture_output=True,text=True,check=True).stdout.strip()
old=json.loads((bench/'Sources/TensorStableCompute/Baselines/2026-10-04-macos-arm64-adam-warm-moments/Configuration.json').read_text())
reference=root/'.silex/tensor-neural-adam-warm-release'
candidate=root/'.silex/tensor-neural-profile-release'
assert sha(reference)==old['binaries']['candidate']
assert sha(root/'Silex/Toolchain/zig-out/bin/silex')==old['compiler_sha256']
files=sorted((root/'Packages/Tensor/Module').rglob('*'))+[bench/'Sources/TensorStableCompute/Neural.sx']
source_hashes={str(p.relative_to(root)):sha(p) for p in files if p.is_file()}
configuration={'date':datetime.now(timezone.utc).isoformat(),'baseline':old['candidate_heads'],'reference_tensor_module_commit':'e742c9a1f55d3dcb4a3d64199607f023e512a193','accepted_tensor_baseline':'27f6cae59223e5cc19483d90f3300337a33cbe40','candidate_source_hashes':source_hashes,'candidate_state':'uncommitted qualified Task-02 sources; exact commit recorded after campaign','compiler_sha256':old['compiler_sha256'],'os':cmd(['sw_vers','-buildVersion']),'cpu':cmd(['sysctl','-n','machdep.cpu.brand_string']),'binaries':{'reference':sha(reference),'candidate':sha(candidate)},'pairs':5,'variants':'reference,off,on alternated; control A/A off','families':'rotated mlp,cnn,rnn','hardware':old['reference']}
assert configuration['cpu']==old['reference']['cpu']
assert configuration['os'] in old['reference']['os']
(res/'Configuration.json').write_text(json.dumps(configuration,ensure_ascii=False,indent=2)+'\n')
for variant in ['reference','off','on','control-a','control-b']:(res/variant).mkdir()
families=('mlp','cnn','rnn')
for rep in range(1,6):
 offset=(rep-1)%3
 for family in families[offset:]+families[:offset]:
  order=('reference','off','on') if rep%2 else ('on','off','reference')
  for variant in order:
   print('MEASURE',rep,family,variant,flush=True)
   binary=reference if variant=='reference' else candidate
   flag='--profile-' if variant=='on' else '--measure-'
   with (res/variant/f'{family}-{rep}.log').open('w') as out,(res/variant/f'{family}-{rep}.time').open('w') as err:
    run=subprocess.run(['/usr/bin/time','-l',str(binary),flag+family],stdout=out,stderr=err)
   if run.returncode:raise SystemExit(f'Invalid sample {rep} {family} {variant}: {run.returncode}')
for rep in range(1,6):
 offset=(rep-1)%3
 for family in families[offset:]+families[:offset]:
  for variant in (('control-a','control-b') if rep%2 else ('control-b','control-a')):
   print('CONTROL',rep,family,variant,flush=True)
   with (res/variant/f'{family}-{rep}.log').open('w') as out,(res/variant/f'{family}-{rep}.time').open('w') as err:
    run=subprocess.run(['/usr/bin/time','-l',str(candidate),'--measure-'+family],stdout=out,stderr=err)
   if run.returncode:raise SystemExit('Invalid A/A control')
assert sha(reference)==configuration['binaries']['reference'] and sha(candidate)==configuration['binaries']['candidate']
assert source_hashes=={str(p.relative_to(root)):sha(p) for p in files if p.is_file()}
print('COMPLETE',res,flush=True)
