from pathlib import Path
import csv, hashlib, json, os, re, statistics, subprocess

w = Path.cwd()
a = w.parent / 'Memory/Artifacts'
out = a / 'Task16-Training-Public-Tracking-Campaign'
out.mkdir(exist_ok=False)
binary = w / 'Qualification/Task16TrainingMeasure-Public-Release'
cli = w / 'Silex/Toolchain/zig-out/bin/silex'
env = dict(os.environ, ZIG_GLOBAL_CACHE_DIR=str(w / '.silex/zig-global'))
sources = {}
for path in sorted((w / 'Packages/Tensor/Module').rglob('*.sx')):
    sources[str(path.relative_to(w/'Packages/Tensor'))] = hashlib.sha256(path.read_bytes()).hexdigest()
benchmark=w/'Silex-Benchmarks/Sources/TensorStableCompute/TrainingStep.sx'
config = dict(benchmark_sha256=hashlib.sha256(benchmark.read_bytes()).hexdigest(), binary_sha256=hashlib.sha256(binary.read_bytes()).hexdigest(), compiler_sha256=hashlib.sha256(cli.read_bytes()).hexdigest(), source_sha256=sources,
              tensor_baseline=subprocess.check_output(['git','rev-parse','HEAD'],cwd=w/'Packages/Tensor',text=True).strip(),
              os=subprocess.check_output(['sw_vers'],text=True).strip(), architecture=subprocess.check_output(['uname','-m'],text=True).strip(),
              hardware=subprocess.check_output(['sysctl','-n','machdep.cpu.brand_string'],text=True).strip(), pairs=16, identical_pairs_per_arm=8,
              gate=dict(minimum_percent=5, mad_multiplier=3, favourable_pairs=14), scope='public TrainingStep MSE/AdamW, native ARM64 macOS Metal only')
(out / 'Config.json').write_text(json.dumps(config,indent=2)+'\n')
(out / 'Protocol.md').write_bytes((w.parent/'Memory/Task16-Training-Tracking-Protocol.md').read_bytes())
rows = []
pattern = re.compile(r'TRAINING_PLAN_MEASURE status=pass case=(\d+) planned=(true|false) iterations=(\d+) elapsed_ms=([\d.eE+-]+) preparation_ms=([\d.eE+-]+) passes=(\d+) observations=(\d+)')
for case in range(3):
    for series, pairs in (('AB',16), ('AA-eager',8), ('AA-grouped',8)):
        for pair in range(pairs):
            order = ('A','B') if pair % 4 in (0,3) else ('B','A')
            for arm in order:
                grouped = arm == 'B' if series == 'AB' else series == 'AA-grouped'
                args = ['/usr/bin/time','-l',str(binary),f'--case-{case}']
                if grouped: args.append('--planned')
                result = subprocess.run(args,cwd=w,env=env,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=120)
                name = f'case-{case}-{series}-{pair:02d}-{arm}.log'
                (out/name).write_text(result.stdout)
                (out/name.replace('.log','.time')).write_text(result.stderr)
                match = pattern.search(result.stdout)
                if result.returncode or not match: raise SystemExit(f'Failed sample {name}: {result.stdout}')
                c,g,n,t,cold,passes,observations = match.groups()
                assert int(c)==case and (g=='true')==grouped and int(passes)==int(n)*(147 if case==2 else 84)
                values = [line for line in result.stdout.splitlines() if line.startswith('TRAINING_VALUE ')]
                assert len(values)==int(observations)
                if rows:
                    first = next((r for r in rows if r['case']==case), None)
                    if first:
                        expected = [line for line in (out/first['log']).read_text().splitlines() if line.startswith('TRAINING_VALUE ')]
                        assert values==expected, f'Full GPU state changed in {name}'
                rss = int(re.search(r'(\d+)\s+maximum resident set size', result.stderr).group(1))
                rows.append(dict(case=case,series=series,pair=pair,arm=arm,planned=grouped,iterations=int(n),elapsed_ms=float(t),preparation_ms=float(cold),passes=int(passes),observations=int(observations),log=name,maximum_resident_bytes=rss))
                (out/'Samples.json').write_text(json.dumps(rows,indent=2)+'\n')
        print(f'case {case} {series}: {pairs} pairs verified',flush=True)
for relative, expected in sources.items(): assert hashlib.sha256((w/'Packages/Tensor'/relative).read_bytes()).hexdigest()==expected
assert hashlib.sha256(binary.read_bytes()).hexdigest()==config['binary_sha256']
assert hashlib.sha256(benchmark.read_bytes()).hexdigest()==config['benchmark_sha256']
assert hashlib.sha256(cli.read_bytes()).hexdigest()==config['compiler_sha256']
with (out/'Samples.csv').open('w') as f:
    writer=csv.DictWriter(f,fieldnames=list(rows[0]));writer.writeheader();writer.writerows(rows)
def paired(case,series):
    selected=[r for r in rows if r['case']==case and r['series']==series]
    deltas=[]
    for pair in sorted({r['pair'] for r in selected}):
        values={r['arm']:r['elapsed_ms'] for r in selected if r['pair']==pair}
        deltas.append(100*(values['B']/values['A']-1))
    median=statistics.median(deltas);mad=statistics.median(abs(d-median) for d in deltas)
    return dict(median_percent=median,mad_percent=mad,favourable=sum(d<0 for d in deltas),deltas_percent=deltas,
                a_median_ms=statistics.median(r['elapsed_ms'] for r in selected if r['arm']=='A'),b_median_ms=statistics.median(r['elapsed_ms'] for r in selected if r['arm']=='B'))
results=[]
for case in range(3):
    ab=paired(case,'AB');aa=[paired(case,s) for s in ('AA-eager','AA-grouped')]
    valid=all(not(abs(r['median_percent'])>=5 and abs(r['median_percent'])>3*r['mad_percent']) for r in aa)
    signal=valid and ab['median_percent']<=-5 and -ab['median_percent']>3*ab['mad_percent'] and ab['favourable']>=14
    results.append(dict(case=case,local_signal='gain' if signal else 'regression' if valid and ab['median_percent']>=5 and ab['median_percent']>3*ab['mad_percent'] and ab['favourable']<=2 else 'indeterminate',identical_controls_valid=valid,ab=ab,aa=aa))
summary=dict(status='pass',samples=len(rows),observations=sum(r['observations'] for r in rows),results=results,scope=config['scope'])
(out/'Summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary,indent=2),flush=True)
